### 4. YAMNet 추론 스크립트 만들기
```
import argparse
import csv
import os
from typing import List, Tuple

import numpy as np
import pandas as pd
import soundfile as sf
import tensorflow as tf
import tensorflow_hub as hub


YAMNET_HOP_SECONDS = 0.48


def load_class_names() -> List[str]:
    class_map_url = (
        "https://raw.githubusercontent.com/tensorflow/models/master/"
        "research/audioset/yamnet/yamnet_class_map.csv"
    )
    class_map_path = tf.keras.utils.get_file(
        "yamnet_class_map.csv", class_map_url
    )

    names = []
    with open(class_map_path, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            names.append(row["display_name"])
    return names


def sec_to_srt_time(t: float) -> str:
    ms = int(round(t * 1000.0))
    h = ms // 3600000
    ms %= 3600000
    m = ms // 60000
    ms %= 60000
    s = ms // 1000
    ms %= 1000
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"


def write_srt(segments: List[Tuple[float, float, str]], out_path: str) -> None:
    with open(out_path, "w", encoding="utf-8") as f:
        for i, (st, et, text) in enumerate(segments, start=1):
            f.write(f"{i}\n")
            f.write(f"{sec_to_srt_time(st)} --> {sec_to_srt_time(et)}\n")
            f.write(f"{text}\n\n")


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--wav", required=True, help="16kHz mono wav path")
    ap.add_argument("--topk", type=int, default=5)
    ap.add_argument("--threshold", type=float, default=0.2)
    ap.add_argument("--out_csv", default="yamnet_frames.csv")
    ap.add_argument("--out_srt", default="yamnet.srt")
    args = ap.parse_args()

    wav_path = args.wav
    if not os.path.exists(wav_path):
        raise FileNotFoundError(wav_path)

    audio, sr = sf.read(wav_path, dtype="float32")
    if audio.ndim != 1:
        raise ValueError("오디오가 모노가 아닙니다. ffmpeg로 -ac 1 변환을 권장합니다.")
    if sr != 16000:
        raise ValueError("샘플레이트가 16kHz가 아닙니다. ffmpeg로 -ar 16000 변환을 권장합니다.")

    class_names = load_class_names()

    model = hub.load("https://tfhub.dev/google/yamnet/1")
    scores, embeddings, spectrogram = model(audio)
    scores_np = scores.numpy()

    mean_scores = scores_np.mean(axis=0)
    topk_idx = np.argsort(mean_scores)[::-1][: args.topk]
    print("=== Overall Top-K (평균 점수 기준) ===")
    for rank, idx in enumerate(topk_idx, start=1):
        print(f"{rank:02d}. {class_names[idx]}  score={mean_scores[idx]:.3f}")

    frame_top1_idx = scores_np.argmax(axis=1)
    frame_top1_score = scores_np.max(axis=1)

    rows = []
    for i in range(scores_np.shape[0]):
        t = i * YAMNET_HOP_SECONDS
        idx = int(frame_top1_idx[i])
        rows.append(
            {
                "time_sec": t,
                "top1_label": class_names[idx],
                "top1_score": float(frame_top1_score[i]),
            }
        )

    df = pd.DataFrame(rows)
    df.to_csv(args.out_csv, index=False, encoding="utf-8-sig")
    print(f"\n프레임 결과 CSV 저장: {args.out_csv}")

    segments = []
    cur_label = None
    cur_start = None

    for i, r in df.iterrows():
        t = float(r["time_sec"])
        label = str(r["top1_label"])
        score = float(r["top1_score"])

        show_label = label if score >= args.threshold else "Unknown"
        if cur_label is None:
            cur_label = show_label
            cur_start = t
            continue

        if show_label != cur_label:
            segments.append((cur_start, t, f"{cur_label}"))
            cur_label = show_label
            cur_start = t

    if cur_label is not None and cur_start is not None:
        end_t = float(df["time_sec"].iloc[-1]) + YAMNET_HOP_SECONDS
        segments.append((cur_start, end_t, f"{cur_label}"))

    write_srt(segments, args.out_srt)
    print(f"SRT 자막 저장: {args.out_srt}")


if __name__ == "__main__":
    main()
```

### 5. 실행하기
가상환경이 활성화된 상태에서
```
python yamnet_infer.py --wav audio.wav
```
YAMNet으로 했을 때 좋은 점

세팅이 쉽고 빠름: TF Hub 기반이라 실행까지 시간이 적게 듭니다.

프레임 단위 결과가 나옴: “언제 어떤 소리가 났는지”를 타임라인으로 보여주기 좋습니다.

클래스가 많음(AudioSet 기반): 여러 생활 소리를 폭넓게 “대충” 잡아내는 데 유리합니다.

다만, 이 한계는 알고 시작하시면 좋습니다

라벨이 ‘생활 도메인’에 딱 맞지 않을 수 있음
예: “문 닫힘”을 “thump/impact”류로 뭉뚱그려 주거나, “식기 소리”를 다른 금속성 소리로 혼동하는 식.

겹친 소리(동시 발생)에서 불리
분리 모델이 아니라 “어떤 소리가 있는지”를 점수로 추정하는 구조라, 강한 소리에 약한 소리가 묻힐 수 있습니다.

프로젝트 목표가 ‘특정 위험 소리’ 탐지라면
결국에는 YAMNet 라벨을 우리 목적 라벨로 매핑하거나, 후속으로 CLAP(제로샷) / 파인튜닝 같은 단계가 필요해질 가능성이 큽니다.

추천 진행 방식 (YAMNet만으로도 비교가 잘 보이게)

각자 클립 10~30초씩 3개

단일 소리 1개

겹친 소리 1개

위험/경보 계열 1개

결과 포맷을 통일

time(초) / top1~top3 label / score

그리고 영상 위에 top1을 자막처럼 얹거나, 시간축 그래프로 보여주면 “비교”가 한눈에 들어옵니다.

“정답 라벨”은 우리 방식으로

예: “문 열림/닫힘”, “식기”, “물 끓음”, “연기 경보기” 같은 우리 라벨을 정해두고

YAMNet 출력 라벨을 가까운 그룹으로 묶어서 정성/정량 평가를 하면 됩니다.
